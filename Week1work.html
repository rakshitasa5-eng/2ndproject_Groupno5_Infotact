Time-Series Preparation¶

In [134]:
shipped = df[df["OrderStatus"] == "Shipped"].copy()              ## Keep Only Shipped Orders
In [135]:
shipped.set_index("OrderDate", inplace=True)                    ## Set OrderDate as Index
In [136]:
monthly = (shipped.groupby("CategoryName").resample("MS")["OrderItemQuantity"].sum().reset_index())   ## Aggregate Monthly Demand
In [137]:
monthly_wide = monthly.pivot(index="OrderDate",columns="CategoryName",values="OrderItemQuantity")    ## Pivot the Data
In [138]:
monthly_wide.plot(figsize=(12,6))
